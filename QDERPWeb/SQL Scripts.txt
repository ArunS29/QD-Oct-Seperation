 CREATE OR ALTER FUNCTION Aicon.fn_GetDefaultCompanyDetails()
RETURNS TABLE
AS
RETURN
(
    WITH Selector AS
    (
        -- 1) grab the raw setting
        SELECT TOP (1)
            DefaultCompanyId
        FROM Aicon.tbl901CompanyDetails
        -- you could filter here if it’s per-user, per-tenant, etc.
    ),
    NormSelector AS
    (
        -- 2) normalize NULL/empty/1 → 1
        SELECT
            CASE
                WHEN DefaultCompanyId IS NULL
                  OR DefaultCompanyId = 1 THEN 1
                ELSE DefaultCompanyId
            END AS sel
        FROM Selector
    ),
    OrderedCompanies AS
    (
        -- 3) number your rows in natural order
        SELECT
            CompanyId,
            DefaultCompanyId,
            currencytype,
            ROW_NUMBER() OVER (ORDER BY CompanyId) AS rn
        FROM Aicon.tbl901CompanyDetails
    )
    -- 4) pick out exactly the Nth row
    SELECT
        oc.CompanyId,
        oc.DefaultCompanyId,
        oc.currencytype
    FROM OrderedCompanies AS oc
    CROSS JOIN NormSelector    AS ns
    WHERE oc.rn = ns.sel
);
GO

 ALTER PROCEDURE [Aicon].[sp901_01UpdateLayout]      
(      
    @LayoutXML NVARCHAR(MAX),    
    @FormID VARCHAR(MAX),    
    @UserID VARCHAR(50),    
    @IsFromWeb BIT = 1  -- Default: TRUE  
)      
AS      
BEGIN  
    SET NOCOUNT ON;  
    IF EXISTS (  
        SELECT 1   
        FROM Aicon.tbl90111LayoutMaster   
        WHERE FormID = @FormID AND UserID = @UserID  
    )  
    BEGIN  
        -- Update existing layout  
        UPDATE Aicon.tbl90111LayoutMaster  
        SET LayoutXml = @LayoutXML  
        WHERE FormID = @FormID AND UserID = @UserID;  
    END  
    ELSE  
    BEGIN  
        -- Insert new layout record  
        INSERT INTO Aicon.tbl90111LayoutMaster (FormID, UserID, LayoutXml)  
        VALUES (@FormID, @UserID, @LayoutXML);  
    END  
END

-- Procedure to calculate total Supplier Outstanding 

CREATE PROCEDURE Aicon.sp_01totSupOutstanding 

AS 

BEGIN 

    DELETE FROM Aicon.tbl01supplieroutstanding

ALTER PROCEDURE [Aicon].[StProAccountLedger]    

    @ParamAccountNo VARCHAR(50),    

    @StartDate DATETIME,   

    @EndDate DATETIME   

AS   

BEGIN   

    SET NOCOUNT ON;   
 
    DECLARE  

        @BaseCurrencyCode VARCHAR(10), 

        @BaseCurrencyImage NVARCHAR(MAX), 

        @BaseCurrencySymbole VARCHAR(50), 

        @BaseCurrencyNote VARCHAR(50), 

        @BaseCurrencyDisplayCode VARCHAR(50);
 
    -- Get Base Currency Code from Company Table 

  
    SELECT  @BaseCurrencyCode = CurrencyType       
  
    FROM Aicon.fn_GetDefaultCompanyDetails();    
 
    -- Get Base Currency Details 

    SELECT TOP 1 

        @BaseCurrencyImage = CurrencyImage, 

        @BaseCurrencySymbole = CurrencySymbole, 

        @BaseCurrencyNote = CurrencyNote, 

        @BaseCurrencyDisplayCode = CurrencyCode 

    FROM tbl20169CurrencyExchange  

    WHERE CurrencyMasterCode = @BaseCurrencyCode;
 
    -- Opening Balance Section (no CurrencyId for historical opening, so CurrencyName not applicable)

    
    SELECT '.Opening Balance' AS VoucherNo, @StartDate AS VoucherDate, '' AS VoucherRefNo, 'Opening Balance' AS VoucherNarration, '' AS VoucherEnteredBy, '' AS VoucherEnteredOn, '' AS VoucherVerifiedBy, 
	'' AS VoucherVerifiedOn, '' AS VoucherApprovedBy, '' AS VoucherApprovedOn, 0 AS VoucherEntryNo, AccountHead, AccountHeadName, '' AS DrCr, 0 AS DrAmount, 0 AS CrAmount, SUM(VoucherAmountFormatted) AS VoucherAmountFormatted,
	SUM(VoucherAmountFormatted * ISNULL(CurrencyRate, 1)) AS ConvertedAmount, 0 AS ConvertedDrAmount, 0 AS ConvertedCrAmount, 'Opening Balance' AS EntryNarration, AccountGroup, MasterGroup, 'Opening Balance' AS VoucherType,
	'Opening Balance' AS SysRemarks, '' AS VoucherModifiedBy, '' AS VoucherModifiedOn, '' AS BillNo, '' AS BillDate, '' AS BillPaidTo, '' AS BillRemarks, @StartDate AS VoucherEffectiveDate, AccountHeadArabic, 
	'' AS ReferenceNote, @BaseCurrencyCode AS CurrencyType, 1 AS ExchangeRate, @BaseCurrencyImage AS CurrencyImage, @BaseCurrencySymbole AS CurrencySymbole, @BaseCurrencyNote AS CurrencyNote, 
	@BaseCurrencyDisplayCode AS CurrencyCode, NULL AS CurrencyName, NULL AS TransactionCurrencySymbol,NULL AS TransactionCurrencyImage

    FROM Aicon.qry201MainVoucherEntriesWithMaster   

    WHERE AccountHead = @ParamAccountNo  

      AND VoucherDate < @StartDate  

      AND ISNULL(IsCalculateOpeningBalance, 0) = 1   

    GROUP BY  

        AccountHead,  

        AccountHeadName,  

        AccountGroup,  

        MasterGroup,  

        AccountHeadArabic   
 
    UNION ALL   
 
    -- Transaction Records Section

    SELECT q.VoucherNo, q.VoucherDate, q.VoucherRefNo, q.VoucherNarration, q.VoucherEnteredBy, q.VoucherEnteredOn, q.VoucherVerifiedBy, q.VoucherVerifiedOn, q.VoucherApprovedBy, q.VoucherApprovedOn, q.VoucherEntryNo, q.AccountHead, q.AccountHeadName, q.DrCr, q.DrAmount, q.CrAmount, q.VoucherAmountFormatted, q.VoucherAmountFormatted * ISNULL(q.CurrencyRate, 1) AS ConvertedAmount, q.DrAmount * ISNULL(q.CurrencyRate, 1) AS ConvertedDrAmount, q.CrAmount * ISNULL(q.CurrencyRate, 1) AS ConvertedCrAmount, q.EntryNarration, ag.AccountGroup, mg.MasterGroup, q.VoucherType, q.SysRemarks, q.VoucherModifiedBy, q.VoucherModifiedOn, q.BillNo, q.BillDate, q.BillPaidTo, q.BillRemarks, q.VoucherEffectiveDate, q.AccountHeadArabic, q.ReferenceNote, @BaseCurrencyCode AS CurrencyType, q.CurrencyRate AS ExchangeRate, @BaseCurrencyImage AS CurrencyImage, @BaseCurrencySymbole AS CurrencySymbole, @BaseCurrencyNote AS CurrencyNote, @BaseCurrencyDisplayCode AS CurrencyCode, cex.CurrencyName, cex.CurrencySymbole AS TransactionCurrencySymbol,cex.CurrencyImage as TransactionCurrencyImage
 
    FROM Aicon.qry201MainVoucherEntriesWithMaster q 

    INNER JOIN Aicon.tbl201ChartOfAccounts ca ON ca.AccountID = q.AccountHead 

    LEFT JOIN Aicon.tbl201AccountGroups ag ON ca.AccountGroupID = ag.AccountGroupID 

    LEFT JOIN Aicon.tbl201MasterGroup mg ON ag.AccountGroupUnder = mg.MasterGroupID 

    LEFT JOIN Aicon.tbl20169CurrencyExchange cex ON q.CurrencyId = cex.CurrencyExchangeID

    WHERE q.AccountHead = @ParamAccountNo  

      AND q.VoucherDate BETWEEN @StartDate AND DATEADD(SECOND, 86399, @EndDate) 

END

-- Opening Balance
SELECT  
    AccountHead,   
    AccountHeadName,   
    SUM(VoucherAmountFormatted), 
    AccountGroup, 
    MasterGroup, 
    'Opening Balance',  
    AccountHeadArabic, 
    AccountGroupAr, 
    MasterGroupAr,
    MIN(VoucherDate) AS VoucherDate,
    DATENAME(m, MIN(VoucherDate)) + ' ' + CAST(DATEPART(yyyy, MIN(VoucherDate)) AS varchar) AS MonthYear,
    NULL AS DrCr,
    SUM(DrAmount) AS DrAmount,
    SUM(CrAmount) AS CrAmount
FROM Aicon.qry201MainVoucherEntriesWithMaster 
WHERE ISNULL(IsCalculateOpeningBalance, 0) = 1  
GROUP BY 
    AccountHead, 
    AccountHeadName, 
    AccountGroup, 
    MasterGroup,  
    AccountHeadArabic, 
    AccountGroupAr, 
    MasterGroupAr
 
UNION ALL
 
-- Transactions during the period
SELECT  
    q.AccountHead, 
    q.AccountHeadName,   
    q.VoucherAmountFormatted, 
    ag.AccountGroup,  
    mg.MasterGroup,  
    CASE 
        WHEN q.VoucherAmountFormatted > 0 THEN 'Transactions Debit' 
        ELSE 'Transactions Credit' 
    END,   
    q.AccountHeadArabic,  
    q.AccountGroupAr,  
    q.MasterGroupAr,
    q.VoucherDate,
    DATENAME(m, q.VoucherEffectiveDate) + ' ' + CAST(DATEPART(yyyy, q.VoucherEffectiveDate) AS varchar) AS MonthYear,
    q.DrCr,
    q.DrAmount,
    q.CrAmount
FROM Aicon.qry201MainVoucherEntriesWithMaster AS q
    RIGHT OUTER JOIN Aicon.tbl201ChartOfAccounts coa ON coa.AccountID = q.AccountHead
    RIGHT OUTER JOIN Aicon.tbl201AccountGroups ag ON coa.AccountGroupID = ag.AccountGroupID
    RIGHT OUTER JOIN Aicon.tbl201MasterGroup mg ON ag.AccountGroupUnder = mg.MasterGroupID;
