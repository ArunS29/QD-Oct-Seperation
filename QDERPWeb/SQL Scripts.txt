  ALTER PROCEDURE [Aicon].[StProAccountLedger]    

    @ParamAccountNo VARCHAR(50),    

    @StartDate DATETIME,   

    @EndDate DATETIME   

AS   

BEGIN   

    SET NOCOUNT ON;   
 
    DECLARE  

        @BaseCurrencyCode VARCHAR(10), 

        @BaseCurrencyImage NVARCHAR(MAX), 

        @BaseCurrencySymbole VARCHAR(50), 

        @BaseCurrencyNote VARCHAR(50), 

        @BaseCurrencyDisplayCode VARCHAR(50);

        @BaseCurrencyDisplayCode = CurrencyCode 

    FROM tbl20169CurrencyExchange  

    WHERE CurrencyMasterCode = @BaseCurrencyCode;
 
    -- Opening Balance Section (no CurrencyId for historical opening, so CurrencyName not applicable)

    SELECT  

        '.Opening Balance' AS VoucherNo,  

        @StartDate AS VoucherDate,    

        '' AS VoucherRefNo, 

        'Opening Balance' AS VoucherNarration,   

        '' AS VoucherEnteredBy, 

        '' AS VoucherEnteredOn,   

        '' AS VoucherVerifiedBy, 

        '' AS VoucherVerifiedOn,   

        '' AS VoucherApprovedBy, 

        '' AS VoucherApprovedOn,   

        0 AS VoucherEntryNo, 

        AccountHead,    

        AccountHeadName, 

        '' AS DrCr, 

        0 AS DrAmount,    

        0 AS CrAmount,  

        SUM(VoucherAmountFormatted) AS VoucherAmountFormatted, 

        SUM(VoucherAmountFormatted * ISNULL(CurrencyRate, 1)) AS ConvertedAmount, 

        0 AS ConvertedDrAmount, 

        0 AS ConvertedCrAmount,

        @BaseCurrencyCode AS CurrencyType,

        @BaseCurrencyImage AS CurrencyImage, 

        @BaseCurrencySymbole AS CurrencySymbole, 

        @BaseCurrencyNote AS CurrencyNote, 

        @BaseCurrencyDisplayCode AS CurrencyCode,

        NULL AS CurrencyName,

		NULL AS TransactionCurrencySymbol

		 -- No CurrencyName for opening

    FROM Aicon.qry201MainVoucherEntriesWithMaster   

    WHERE AccountHead = @ParamAccountNo  

      AND VoucherDate < @StartDate  

      AND ISNULL(IsCalculateOpeningBalance, 0) = 1   

    GROUP BY  

        AccountHead,  

        AccountHeadName,  

        AccountGroup,  

        MasterGroup,  

        AccountHeadArabic   
 
    UNION ALL   
 
    -- Transaction Records Section

    SELECT      

        q.VoucherNo,  

        q.VoucherDate,    

        q.VoucherRefNo,  

        q.VoucherNarration,    

        q.VoucherEnteredBy,  

        q.VoucherEnteredOn,    

        q.VoucherVerifiedBy,  

        q.VoucherVerifiedOn,    

        q.VoucherApprovedBy,  

        q.VoucherApprovedOn,    

        q.VoucherEntryNo,  

        q.AccountHead,    

        q.AccountHeadName,  

        q.DrCr,  

        q.DrAmount,    

        q.CrAmount,  

        q.VoucherAmountFormatted, 

        q.VoucherAmountFormatted * ISNULL(q.CurrencyRate, 1) AS ConvertedAmount, 

        q.DrAmount * ISNULL(q.CurrencyRate, 1) AS ConvertedDrAmount, 

        q.CrAmount * ISNULL(q.CurrencyRate, 1) AS ConvertedCrAmount, 

        q.EntryNarration,  

        ag.AccountGroup,  

        mg.MasterGroup,   

        q.VoucherType,  

        q.SysRemarks,  

        q.VoucherModifiedBy,  

        q.VoucherModifiedOn,    

        q.BillNo,  

        q.BillDate,  

        q.BillPaidTo,   

        q.BillRemarks,  

        q.VoucherEffectiveDate,  

        q.AccountHeadArabic,  

        q.ReferenceNote, 

        @BaseCurrencyCode AS CurrencyType, 

        q.CurrencyRate AS ExchangeRate, 

        @BaseCurrencyImage AS CurrencyImage, 

        @BaseCurrencySymbole AS CurrencySymbole, 

        @BaseCurrencyNote AS CurrencyNote, 

        @BaseCurrencyDisplayCode AS CurrencyCode,

        cex.CurrencyName,

		cex.CurrencySymbole AS TransactionCurrencySymbol
 
    FROM Aicon.qry201MainVoucherEntriesWithMaster q 

    INNER JOIN Aicon.tbl201ChartOfAccounts ca ON ca.AccountID = q.AccountHead 

    LEFT JOIN Aicon.tbl201AccountGroups ag ON ca.AccountGroupID = ag.AccountGroupID 

    LEFT JOIN Aicon.tbl201MasterGroup mg ON ag.AccountGroupUnder = mg.MasterGroupID 

    LEFT JOIN Aicon.tbl20169CurrencyExchange cex ON q.CurrencyId = cex.CurrencyExchangeID

    WHERE q.AccountHead = @ParamAccountNo  

      AND q.VoucherDate BETWEEN @StartDate AND DATEADD(SECOND, 86399, @EndDate) 

END

    INSERT INTO Aicon.tbl01supplieroutstanding (AccountHead, Balance) 

    SELECT 'supplier outstanding', SUM([Balance]) 

    FROM Aicon.qry20115BillsPayableOutStanding 

    WHERE Balance > 0; 

END; 

  

-- Procedure to calculate total Client Outstanding 

CREATE PROCEDURE sp_01totClientOutstanding 

AS 

BEGIN 

    DELETE FROM Aicon.tbl01supplieroutstanding 

    WHERE AccountHead = 'Client outstanding'; 

  

    INSERT INTO Aicon.tbl01supplieroutstanding (AccountHead, Balance) 

    SELECT 'Client outstanding', SUM(Balance) 

    FROM Aicon.Qry20115BillsOutStanding 

    WHERE Balance > 0; 

END; 

  

-- Procedure to calculate total Cash Balance 

CREATE PROCEDURE sp_01totcashbalance 

AS 

BEGIN 

    DELETE FROM Aicon.tbl01supplieroutstanding 

    WHERE AccountHead = 'Cash Balance'; 

  

    INSERT INTO Aicon.tbl01supplieroutstanding (AccountHead, Balance) 

    SELECT 'Cash balance', SUM(ISNULL(DrAmount, 0) - ISNULL(CrAmount, 0)) AS CashBalance 

    FROM Aicon.Qry201MainVoucherEntriesWithMaster 

    WHERE AccountGroup IN ('BANK ACCOUNTS', 'CASH-IN-HAND'); 

END;
