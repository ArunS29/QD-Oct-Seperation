CREATE   PROCEDURE dbo.StProTrialBalance_New  
(  
    @StartDate DATETIME,  
    @EndDate   DATETIME  
)  
AS  
BEGIN  
    SET NOCOUNT ON;  
  
    -----------------------------------------------------------------  
    -- 1. Opening Balance (all data before @StartDate)  
    -----------------------------------------------------------------  
    SELECT  
        '.Opening Balance' AS VoucherNo,  
        @StartDate AS VoucherDate,  
        0 AS VoucherEntryNo,  
        t.AccountHead,  
        t.AccountHeadName,  
        '' AS DrCr,  
        SUM(t.DrAmount) AS DrAmount,  
        SUM(t.CrAmount) AS CrAmount,  
        SUM(t.VoucherAmountFormatted) AS VoucherAmountFormatted,  
        t.AccountGroup,  
        t.MasterGroup,  
        'Opening Balance' AS VoucherType,  
        'Opening Balance' AS Transactions,  
        'Opening Balance' AS MonthYear,  
        0 AS MonthNumber,  
        t.Category,  
        '' AS VoucherRefNo,  
        '' AS VoucherNarration,  
        '' AS EntryNarration,  
        '' AS SysRemarks  
    FROM dbo.TrialBalanceResults t  
    WHERE t.VoucherDate < @StartDate  
        
    GROUP BY t.AccountHead, t.AccountHeadName, t.AccountGroup, t.MasterGroup, t.Category  
  
    UNION ALL  
  
    -----------------------------------------------------------------  
    -- 2. Transactions (between @StartDate and @EndDate)  
    -----------------------------------------------------------------  
    SELECT  
        t.VoucherNo,  
        CAST(t.VoucherDate AS DATE) AS VoucherDate,  
        t.VoucherEntryNo,  
        t.AccountHead,  
        t.AccountHeadName,  
        t.DrCr,  
        t.DrAmount,  
        t.CrAmount,  
        t.VoucherAmountFormatted,  
        t.AccountGroup,  
        t.MasterGroup,  
        t.VoucherType,  
        'Transactions' AS Transactions,  
        DATENAME(m, t.VoucherDate) + ' ' + CAST(YEAR(t.VoucherDate) AS NVARCHAR) AS MonthYear,  
        MONTH(t.VoucherDate) AS MonthNumber,  
        t.Category,  
        t.VoucherRefNo,  
        t.VoucherNarration,  
        t.EntryNarration,  
        t.SysRemarks  
    FROM dbo.TrialBalanceResults t  
    WHERE t.VoucherDate BETWEEN @StartDate AND DATEADD(SECOND, 86399, @EndDate) -- end of day  
  
    ORDER BY AccountHead, VoucherDate, VoucherEntryNo;  
END

          SUM(VoucherAmountFormatted) AS VoucherAmountFormatted, AccountGroup, MasterGroup, 'Opening Balance' AS VoucherType, 'Opening Balance' AS Transactions, 'Opening Balance' AS MonthYear, 0 as MonthNumber,  
   MasterGroupCategory AS Category, '' as VoucherRefNo, '' as VoucherNarration, '' as EntryNarration, '' as SysRemarks  
   FROM         dbo.qry201MainVoucherEntriesWithMaster  
   WHERE     (VoucherDate < @StartDate) AND (ISNULL(IsCalculateOpeningBalance, 0) = 1)  
   GROUP BY AccountHead, AccountHeadName, AccountGroup, MasterGroup, MasterGroupCategory  
   UNION ALL  
   SELECT     qry201MainVoucherEntriesWithMaster_1.VoucherNo, DATEADD(dd, DATEDIFF(dd, 0, qry201MainVoucherEntriesWithMaster_1.VoucherDate), 0) as VoucherDate, qry201MainVoucherEntriesWithMaster_1.VoucherEntryNo,   
          qry201MainVoucherEntriesWithMaster_1.AccountHead, qry201MainVoucherEntriesWithMaster_1.AccountHeadName, qry201MainVoucherEntriesWithMaster_1.DrCr,   
          qry201MainVoucherEntriesWithMaster_1.DrAmount, qry201MainVoucherEntriesWithMaster_1.CrAmount, qry201MainVoucherEntriesWithMaster_1.VoucherAmountFormatted,   
          dbo.tbl201AccountGroups.AccountGroup, dbo.tbl201MasterGroup.MasterGroup, qry201MainVoucherEntriesWithMaster_1.VoucherType, 'Transactions' AS Transactions,   
    DATENAME(m, qry201MainVoucherEntriesWithMaster_1.VoucherDate) + ' ' + CAST(DATEPART(yyyy, qry201MainVoucherEntriesWithMaster_1.VoucherDate) AS varchar) AS MonthYear, DATEPART(mm, VoucherDate) as MonthNumber  
   , dbo.tbl201MasterGroup.MasterGroupCategory, VoucherRefNo, VoucherNarration, EntryNarration, SysRemarks  
   FROM         dbo.tbl201MasterGroup RIGHT OUTER JOIN  
          dbo.tbl201AccountGroups ON dbo.tbl201MasterGroup.MasterGroupID = dbo.tbl201AccountGroups.AccountGroupUnder RIGHT OUTER JOIN  
          dbo.tbl201ChartOfAccounts ON dbo.tbl201AccountGroups.AccountGroupID = dbo.tbl201ChartOfAccounts.AccountGroupID RIGHT OUTER JOIN  
          dbo.qry201MainVoucherEntriesWithMaster AS qry201MainVoucherEntriesWithMaster_1 ON dbo.tbl201ChartOfAccounts.AccountID = qry201MainVoucherEntriesWithMaster_1.AccountHead  
   WHERE     (qry201MainVoucherEntriesWithMaster_1.VoucherDate BETWEEN @StartDate AND DATEADD(hh, 23, DATEADD(mi, 59, DATEADD(ss, 59, @EndDate))));  
  
 END

   SELECT     '.Opening Balance' AS VoucherNo, @StartDate AS VoucherDate, 0 AS VoucherEntryNo, AccountHead, AccountHeadName, '' AS DrCr, SUM(DrAmount) AS DrAmount, SUM(CrAmount) AS CrAmount,  
          SUM(VoucherAmountFormatted) AS VoucherAmountFormatted, AccountGroup, MasterGroup, 'Opening Balance' AS VoucherType, 'Opening Balance' AS Transactions, 'Opening Balance' AS MonthYear, 0 as MonthNumber,  
   'Balance Sheet' AS Category, '' as VoucherRefNo, '' as VoucherNarration, '' as EntryNarration, '' as SysRemarks

   WHERE     (VoucherEffectiveDate < @StartDate) AND (ISNULL(IsCalculateOpeningBalance, 0) = 1)  
   GROUP BY AccountHead, AccountHeadName, AccountGroup, MasterGroup  
   UNION ALL

          qry201MainVoucherEntriesWithMaster_1.DrAmount, qry201MainVoucherEntriesWithMaster_1.CrAmount, qry201MainVoucherEntriesWithMaster_1.VoucherAmountFormatted,   
          dbo.tbl201AccountGroups.AccountGroup, dbo.tbl201MasterGroup.MasterGroup, qry201MainVoucherEntriesWithMaster_1.VoucherType, 'Transactions' AS Transactions,   
    DATENAME(m, qry201MainVoucherEntriesWithMaster_1.VoucherEffectiveDate) + ' ' + CAST(DATEPART(yyyy, qry201MainVoucherEntriesWithMaster_1.VoucherEffectiveDate) AS varchar) AS MonthYear, DATEPART(mm, VoucherEffectiveDate) as MonthNumber  
   , dbo.tbl201MasterGroup.MasterGroupCategory, VoucherRefNo, VoucherNarration, EntryNarration, SysRemarks  
   FROM         dbo.tbl201MasterGroup RIGHT OUTER JOIN  
          dbo.tbl201AccountGroups ON dbo.tbl201MasterGroup.MasterGroupID = dbo.tbl201AccountGroups.AccountGroupUnder RIGHT OUTER JOIN  
          dbo.tbl201ChartOfAccounts ON dbo.tbl201AccountGroups.AccountGroupID = dbo.tbl201ChartOfAccounts.AccountGroupID RIGHT OUTER JOIN  
          dbo.qry201MainVoucherEntriesWithMaster AS qry201MainVoucherEntriesWithMaster_1 ON dbo.tbl201ChartOfAccounts.AccountID = qry201MainVoucherEntriesWithMaster_1.AccountHead  
   WHERE     (qry201MainVoucherEntriesWithMaster_1.VoucherEffectiveDate BETWEEN @StartDate AND DATEADD(hh, 23, DATEADD(mi, 59, DATEADD(ss, 59, @EndDate))));  
  
 END
